Setup Supabase

Supabase sudah diisi dari kode. Form ini tetap tersedia bila nanti project URL atau publishable key berubah.

SQL untuk Supabase

create table if not exists public.price_uploads (
  id uuid primary key default gen_random_uuid(),
  uploaded_at timestamptz not null default now(),
  uploaded_by text,
  filename text
);

create table if not exists public.price_items (
  id bigserial primary key,
  upload_id uuid references public.price_uploads(id) on delete cascade,
  code text,
  name text not null,
  stock_text text,
  stock_number numeric,
  price_pcs numeric,
  price_dus numeric,
  price_pak numeric,
  promos jsonb default '{}'::jsonb,
  note text,
  created_at timestamptz not null default now()
);

alter table public.price_uploads enable row level security;
alter table public.price_items enable row level security;

create policy "public read uploads" on public.price_uploads for select using (true);
create policy "public read items" on public.price_items for select using (true);
create policy "admin insert uploads" on public.price_uploads for insert to authenticated with check (true);
create policy "admin insert items" on public.price_items for insert to authenticated with check (true);
create policy "admin delete uploads" on public.price_uploads for delete to authenticated using (true);
create policy "admin delete items" on public.price_items for delete to authenticated using (true);

grant select on public.price_uploads to anon;
grant select on public.price_items to anon;
grant select, insert, delete on public.price_uploads to authenticated;
grant select, insert, delete on public.price_items to authenticated;
grant usage, select on sequence public.price_items_id_seq to authenticated;
